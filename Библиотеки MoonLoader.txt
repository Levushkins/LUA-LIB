Представляю вам MoonMonet.

MoonMonet — это инновационная библиотека для GTA SA, которая генерирует три цветовых акцента и два нейтральных цветовых акцента на основе указанного основного цвета. Она работает по аналогии с системой «Monet» в Android 12, которая была представлена вместе с концепцией «Material You». Изначально библиотека была написана на языке Kotlin и применялась для реализации системы «Monet» в бета-версиях Android 12 (проект android12-extensions). Недавно я решил реализовать подобный функционал для SAMP и переписал библиотеку на Rust, после чего внедрил её в проект Police Helper | «Reborn». Результат превзошел ожидания: система цветов получила исключительно положительные отзывы от всех, кого я просил оценить визуальную составляющую.

GitHub проекта: Northn/MoonMonet
GitHub MonetRs: Northn/monet-rs
Версия MoonMonet: 0.1.0; состояние: Stable

Установка: папку MoonMonet из архива MoonMonet0.1.0.zip, приложенного к теме, необходимо перетащить в каталог: *Корневая папка с игрой*/moonloader/lib.

Применение

Библиотека включает в себя всего одну функцию — buildColors, которая принимает три параметра: основной цвет, коэффициент яркости и режим работы. Благодаря этому использование инструмента становится максимально простым.

Lua:
local COLOR = 0xff00babe; -- Цвет в формате AARRGGBB
local CHROMA_MULTIPLIER = 1.0; -- Яркость цвета. Рекомендуемое значение: от 0.5 до 2.0
local colors = MoonMonet.buildColors(COLOR, CHROMA_MULTIPLIER, true);

В результате выполнения функции мы получаем структуру, состоящую из вложенных структур: Colors и SubColors.
Структура Colors включает в себя наборы SubColors: accent1, accent2, accent3 (основные акценты) и neutral1, neutral2 (нейтральные оттенки).
Каждый набор SubColors содержит ряд цветов в формате AARRGGBB:
--[[    
    color_0;    color_10;   color_50;
    color_100;  color_200;  color_300;
    color_400;  color_500;  color_600;
    color_700;  color_800;  color_900;
    color_1000;
]]

Примеры обращения к сгенерированным цветам:
local generated_color = colors.accent1.color_50;
local generated_color = colors.accent2.color_100;
local generated_color = colors.accent3.color_200;
local generated_color = colors.neutral1.color_300;
local generated_color = colors.neutral2.color_400;
-- И так далее.

Согласитесь, это очень легко. При этом польза от библиотеки при должной фантазии может быть огромной!

Вспомогательная функция для удобной конвертации цветов

Поскольку я являюсь сторонником красивого и качественного кода, я подготовил функцию, которая позволяет с удобством конвертировать цвета в необходимые форматы.

Lua:
do
    local bit = require 'bit'

    local function join_argb(a, r, g, b)
        local argb = b  -- b
        argb = bit.bor(argb, bit.lshift(g, 8))  -- g
        argb = bit.bor(argb, bit.lshift(r, 16)) -- r
        argb = bit.bor(argb, bit.lshift(a, 24)) -- a
        return argb
    end
 
    local function explode_argb(argb)
        local a = bit.band(bit.rshift(argb, 24), 0xFF)
        local r = bit.band(bit.rshift(argb, 16), 0xFF)
        local g = bit.band(bit.rshift(argb, 8), 0xFF)
        local b = bit.band(argb, 0xFF)
        return a, r, g, b
    end

    local function ARGBtoRGB(color)
        return bit.band(color, 0xFFFFFF)
    end

    function ColorAccentsAdapter(color)
        local a, r, g, b = explode_argb(color)

        local ret = {a = a, r = r, g = g, b = b}

        function ret:apply_alpha(alpha)
            self.a = alpha
            return self
        end

        function ret:as_u32()
            return join_argb(self.a, self.b, self.g, self.r)
        end

        function ret:as_vec4()
            return imgui.ImVec4(self.r / 255, self.g / 255, self.b / 255, self.a / 255)
        end

        function ret:as_argb()
            return join_argb(self.a, self.r, self.g, self.b)
        end

        function ret:as_rgba()
            return join_argb(self.r, self.g, self.b, self.a)
        end

        function ret:as_chat()
            return string.format("%06X", ARGBtoRGB(join_argb(self.a, self.r, self.g, self.b)))
        end
     
        return ret
    end
end

Пример использования конвертера

local CHROMA_MULTIPLIER = 1.0;

-- Вывод сообщения в чат с использованием сгенерированного цвета

Что такое SNET и для чего его можно использовать?

SNET (Simple Network) — это модуль сетевого интерфейса, работающий на основе библиотеки LuaSocket и использующий в своей работе протокол UDP (User Datagram Protocol), который чаще всего применяется при реализации сетевых игр. Помимо стандартного функционала UDP, в модуль были добавлены некоторые функции, которые присущи TCP (Transmission Control Protocol), но изначально отсутствуют в UDP. Так, например, SNET имеет функционал, позволяющий убедиться в том, что тот или иной пакет был доставлен до получателя, а также систему приоритетов отправки пакетов. 

Чем выше приоритет, тем больше попыток сделает ваш сервер или клиент, чтобы отправить пакет в случае, если вторая сторона не отчиталась о его получении. Таким образом, на модуле SNET можно писать как большие проекты, такие как, например, SL:MP — прародитель SNET (именно там впервые была использована эта технология), так и маленькие проекты: чаты, мини-игры по вашему желанию (крестики-нолики, шашки, шахматы по сети) и прочее, что взбредет вам в голову, ведь функционал не ограничен.

Есть ли какие-нибудь ограничения по количеству клиентов?

Нет, абсолютно никаких ограничений нет, ибо UDP как протокол не имеет функции подключения. Вы можете присоединить столько клиентов, сколько хотите — все эти операции происходят при помощи обмена пакетами. Простой пример: клиент отправляет пакет с запросом на подключение, сервер проверяет, не зарегистрирован ли ещё этот клиент в системе, и если нет, то происходит регистрация, а после клиенту отправляется пакет о том, что он зарегистрирован.

Подходит ли модуль для каких-либо площадок, кроме MoonLoader'a?

Да, SNET без каких-либо проблем запускается при помощи LuaJIT вне игры, а тот, как известно, поддерживается и ОС Windows, и различными сборками Linux. Соответственно, можно собрать сервер и поставить его на VPS/VDS — нет никаких ограничений в этом плане. При этом SNET полностью совместим с MoonLoader`ом.

Примеры использования SNET в своих проектах

Ниже представлены две версии кода: один для сервера, второй для клиента — вместе они работают отлично и обмениваются пакетами. Обратите внимание на комментарии: в них содержится полезная информация, которая может вам пригодиться при написании своего клиента и/или сервера.

Client Example:
-- Пример использования SL:NET в качестве клиента
-- Примечание: SLNetLoop рекомендуется выполнять асинхронно, если возможно
-- Можно также выполнять в основном цикле, но возможны некоторые задержки

-- Нужно для корректной работы примера, можно убрать, если есть папка с библиотеками
package.path = package.path .. ';../?.lua;../?.luac;../../?.lua;../../?.luac'
package.cpath = package.cpath .. ';../?.dll;../../?.dll'
-- Нужно для корректной работы примера, можно убрать, если есть папка с библиотеками

require 'SLNet' -- подключаем модуль SL:NET для дальнейшего использования

netHandle = SLNetInit() -- инициализируем модуль SL:Net, нужно делать при каждой смене сервера
-- то есть, если вы хотите подключиться к другому серверу, сначала инициализируйте SL:NET
netHandle:connect('127.0.0.1', 6666) -- подключаемся к серверу по набору IP:PORT
-- можно также использовать функцию SLNetConnect(netHandle, IP, Port)
netHandle:setPrefix('EXMPL') -- устанавливает проверочный префикс в начало пакета
-- сервер и все клиенты должны иметь один и тот же префикс, иначе пакет не обработается
-- можно также использовать функцию SLNetSetPrefix(netHandle, Prefix)


-- в данной callback-функции мы будем обрабатывать приходящие пакеты
function onReceiveData(packetID, bitStream)
  print('New Incoming Message => ' .. bitStream:export())

  if packetID == 1 then -- если нам пришел первый PacketID
    -- собираем те параметры, которые нужно получить от сервера
    local a, b = bitStream:read(UINT8), bitStream:read(STRING, bitStream:read(UINT8))
    print('Message Content: ' .. a .. ', ' .. b)
  end

  local testString = 'Hello SL:NET World!'

  local BSNew = BitStream:new() -- создадим BitStream и запишем в него данные
  BSNew:write(UINT8, 128):write(UINT8, #testString):write(STRING, testString)

  SLNetSend(netHandle, 1, BSNew, 5) -- отправляем сообщение с PacketID 1 и приоритетом 5
  -- приоритет помогает увеличить шанс доставки пакета, по сути он будет отправляться до
  -- тех пор, пока не придет подтверждение получения, а количество повторов зависит от
  -- значения четвертого аргумента, не может быть больше 255 повторений — это лимит
  -- пакеты, не имеющие сильного значения, лучше отправлять с приоритетом ноль
end

netHandle:setHook(onReceiveData) -- ставим хук на входящие пакеты
-- можно также использовать функцию SLNetSetHook(netHandle, Callback)

SLNetSend(netHandle, 1, nil, 5) -- отправить можно и пустой пакет, зачем — не знаю
while true do
  netHandle:loop() -- обязательно вызывать в цикле, желательно в начале
  -- этот метод отвечает за прогон основных потоков получения и отправки данных
  -- вызывать только после того, как произошел bind/connect, иначе вернет FALSE
  -- метод также имеет зеркальную функцию: SLNetLoop(netHandle)
end

-- Пример использования SL:NET в качестве сервера
-- Примечание: SLNetLoop рекомендуется выполнять асинхронно, если возможно
-- Можно также выполнять в основном цикле, но возможны некоторые задержки

-- Нужно для корректной работы примера, можно убрать, если есть папка с библиотеками
package.path = package.path .. ';../?.lua;../?.luac;../../?.lua;../../?.luac'
package.cpath = package.cpath .. ';../?.dll;../../?.dll'
-- Нужно для корректной работы примера, можно убрать, если есть папка с библиотеками

require 'SLNet'

netHandle = SLNetInit() -- инициализируем модуль SL:Net, вызывать при каждой смене IP:PORT
netHandle:bind('*', 6666) -- занимаем локальный IP и порт 6666, на него будут подключаться клиенты
-- можно также использовать функцию SLNetBind(netHandle, IP, Port)
netHandle:setPrefix('EXMPL') -- устанавливает проверочный префикс в начало пакета
-- сервер и все клиенты должны иметь один и тот же префикс, иначе пакет не обработается
-- можно также использовать функцию SLNetSetPrefix(netHandle, Prefix)

-- в данной callback-функции мы будем обрабатывать приходящие пакеты
function onReceiveData(packetID, bitStream, addr, port)
  print('Message from ' .. addr .. ':' .. port)

  if packetID == 1 then -- если нам пришел первый PacketID
    -- собираем те параметры, которые нужно получить от сервера
    local a, b = bitStream:read(UINT8), bitStream:read(STRING, bitStream:read(UINT8))
    print('Message Content: ' .. a .. ', ' .. b)
  end

  local testString = 'Hello SL:NET World!'

  local BSNew = BitStream:new() -- создадим BitStream и запишем в него данные
  BSNew:write(UINT8, 128):write(UINT8, #testString):write(STRING, testString)

  SLNetSend(netHandle, 1, BSNew, addr, port, 5) -- отправляем сообщение с PacketID 1 и приоритетом 5
  -- приоритет помогает увеличить шанс доставки пакета, по сути он будет отправляться до
  -- тех пор, пока не придет подтверждение получения, а количество повторов зависит от
  -- значения четвертого аргумента, не может быть больше 255 повторений — это лимит
  -- пакеты, не имеющие сильного значения, лучше отправлять с приоритетом ноль
end

netHandle:setHook(onReceiveData) -- ставим хук на входящие пакеты
-- можно также использовать функцию SLNetSetHook(netHandle, Callback)

while true do
  netHandle:loop() -- обязательно вызывать в цикле, желательно в начале
  -- этот метод отвечает за прогон основных потоков получения и отправки данных
  -- вызывать только после того, как произошел bind/connect, иначе вернет FALSE
  -- метод также имеет зеркальную функцию: SLNetLoop(netHandle)
end

MoonBot — это библиотека для MoonLoader, позволяющая создавать внутриигровых ботов, подобных тем, что вы могли видеть в RakSAMP, RakBot, Overlight и других инструментах. У каждого бота имеется свой индекс, по которому к нему можно обращаться. Важно отметить, что индексы не являются общими для всех скриптов: у каждого отдельного скрипта может быть бот с первым индексом, но доступа к чужому боту с таким же индексом у него нет.

Примеры возможностей и начало работы:
Для удобства в дальнейшем изложении будем считать, что переменная mb инициализирована следующим образом:
local mb = require('MoonBot') -- Имя аргумента может быть любым, далее используется 'mb'.

Методы, вызываемые из основного модуля (mb):
Примечание: Аргументы в угловых скобках <> являются опциональными, в квадратных [] — обязательными.

1. local bot = mb.add(<Имя>) — Регистрирует нового бота и возвращает его хендл.
2. mb.remove([Индекс бота]) — Удаляет бота по его индексу.
3. mb.disconnectAfterUnload([bool status]) — Устанавливает, будут ли боты автоматически удаляться при выгрузке скрипта.
4. local bots = mb.getBots() — Возвращает таблицу с хендлами всех активных ботов.
5. local bot = mb.getBotHandleByIndex([Индекс бота]) — Возвращает хендл конкретного бота по его индексу.
6. local bitStream = mb.getBitStream() — Создает и возвращает объект BitStream. Требует ручного удаления методом bs:remove().
7. local data = mb.getPlayerData() — Возвращает объект данных игрока (PlayerData). Удаляется автоматически.
8. local data = mb.getIncarData() — Возвращает объект данных игрока в авто (IncarData). Удаляется автоматически.
9. local data = mb.getPassengerData() — Возвращает объект данных пассажира (PassengerData). Удаляется автоматически.
10. local data = mb.getUnoccupiedData() — Возвращает объект данных незанятого транспорта (UnoccupiedData). Удаляется автоматически.
11. local data = mb.getTrailerData() — Возвращает данные прицепа (TrailerData). Удаляется автоматически.
12. local data = mb.getBulletData() — Возвращает данные пуль (BulletData). Удаляется автоматически.
13. local data = mb.getSpectatorData() — Возвращает данные наблюдателя (SpectatorData). Удаляется автоматически.
14. local data = mb.getAimData() — Возвращает данные прицеливания (AimData). Удаляется автоматически.

Методы и свойства хендла бота (bot):
— bot.index: Индекс бота (только чтение).

— bot:sendSpectatorData([data]): Отправляет пакет синхронизации ID_SPECTATOR_SYNC.
— bot:sendBulletData([data]): Отправляет пакет синхронизации ID_BULLET_SYNC.
— bot:sendAimData([data]): Отправляет пакет синхронизации ID_AIM_SYNC.

— bot:sendChat([Message]): Отправляет сообщение в общий чат.

— bot:sendInteriorChange([interiorId]): Уведомляет сервер о смене интерьера.

— bot:useProxy([bool], function(state) end): Включает или выключает использование прокси.

Методы объекта BitStream (bs):
— bs:remove(): Удаляет объект битстрима (обязательно для объектов, созданных через mb.getBitStream()).
— bs:setReadOffset([offset]): Устанавливает смещение чтения в битах.
— bs:setWriteOffset([offset]): Устанавливает смещение записи в битах.
— bs:resetReadPointer() / bs:resetWritePointer(): Сброс указателей.
— bs:ignoreBits([bits]): Пропуск указанного количества битов.
— bs:getNumberOfBitsUsed(): Общее количество записанных битов.
— bs:getNumberOfUnreadBits(): Количество оставшихся непрочитанных битов.
— Методы чтения: readBool, readInt8, readInt16, readInt32, readFloat, readString8, readString16, readString32.
— Методы записи: writeBool, writeInt8, writeInt16, writeInt32, writeFloat, writeString8, writeString16, writeString32.
— bs:decodeString([maxChars]): Декодирует строку из битстрима.

Структуры пакетов:

    -- Обработка входящих пакетов.

Пример реализации команд для добавления и удаления бота:
function main()
    sampRegisterChatCommand('bot.add', addCommand)
    sampRegisterChatCommand('bot.remove', removeCommand)
    wait(-1)
end

function addCommand(arg)
    local bot = mb.add(arg) -- Если ник пустой, будет Unnamed_Index
    bot:connect() -- Подключение к текущему серверу
    sampAddChatMessage(string.format('Connecting %s', bot.name), -1)
end

function removeCommand(arg)
    local index = tonumber(arg)
    if index then
        local bot = mb.getBotHandleByIndex(index)
        if bot ~= nil then
            sampAddChatMessage(string.format('Deleting %s', bot.name), -1)
            mb.remove(index)
        end
    end
end

Lua vmt+jmp/call hook | Библиотека hooks.lua

Здрасте, продолжаем насиловать игру хуками, только уже виртуальных таблиц (обычный jmp hook обсуждался ранее). Решил сделать библиотеку, в которой реализованы и этот, и jmp хук. Файл библиотеки необходимо кинуть в папку /moonloader/lib/. Затрагивая jmp hook, стоит отметить, что использование практически не изменилось, только теперь мы пишем через пространство имен:

Lua:
local hook = require("hooks")
tableHook = hook.jmp.new(прототип, калбэк, адрес, размер)
-- вместо прежнего tableHook = hook.new(прототип, калбэк, адрес, размер)

Все хуки при завершении работы скрипта будут сняты автоматически. Теперь перейдем к vmt hook и особенностям его использования.

Требования:
1. Знание FFI.
2. Знание прототипа метода.
3. Знание индекса метода.

Использование:
Покажу на примере хука метода ракнета, так как изначально под него это чудо и писалось. В примере ниже приведена реализация для SAMP 0.3.7-R1 (хотя изначально делалось под R3). Поскольку мы пишем под ракнет, нам понадобится SF.lua или хотя бы библиотека bitstream.lua.

Первым делом подключаем необходимые библиотеки:

Lua:

local bitstream = require("SFlua.bitstream")

Далее ожидаем инициализации SAMP и находим указатель на RakClientInterface:

Lua:
local samp = getModuleHandle("samp.dll")
local pSAMPInfo = 0
while pSAMPInfo == 0 do wait(0)
    pSAMPInfo = ffi.cast("uintptr_t*", samp + 0x21A0F8)[0]
end
local pRakClient = ffi.cast("intptr_t*", pSAMPInfo + 0x3C9)[0]

Инициализируем хук:

Lua:
local rakClient = hook.vmt.new(pRakClient)
-- Примечание: можно не выпендриваться и использовать sampGetRakclientInterface(), но тогда появится зависимость от SAMPFUNCS.

После инициализации становятся доступны следующие методы:
rakClient.hookMethod(cast, func, method)
rakClient.unHookMethod(method)
rakClient.unHookAll()
rakClient.this

Узнаем прототип и индекс, пишем калбек и ставим хук на функцию отправки пакетов:

Lua:
sendPacket = rakClient.hookMethod("bool(__thiscall*)(void* this, SFL_BitStream* bitStream, char priority, char reliability, char orderingChannel)", sendPacket, 6)
-- Структура SFL_BitStream доступна только если подключена либа bitstream из SF-lua.
-- Функция возвращает нам оригинальный метод.

function sendPacket(this, bitStream, priority, reliability, orderingChannel)
    print("packet id", bitStream.data[0]) -- выводим ID пакета
    -- Оригинальные названия пакетов можно найти в библиотеке sampfuncs.lua.
    local res = sendPacket(this, bitStream, priority, reliability, orderingChannel) -- вызываем оригинал
    return res
end

Полный код примера:

local ffi = require 'ffi'
local bitstream = require("bitstream")
local hook = require("hooks")

function main()

        pSAMPInfo = ffi.cast("uintptr_t*", samp + 0x21A0F8)[0]
    end
    local pRakClient = ffi.cast("intptr_t*", pSAMPInfo + 0x3C9)[0]
    local rakClient = hook.vmt.new(pRakClient)

    -- Доступные методы:
    -- original_method = rakClient.hookMethod(cast, func, method)
    -- rakClient.unHookMethod(method)
    -- rakClient.unHookAll()
    -- rakClient.this

    sendPacket = rakClient.hookMethod("bool(__thiscall*)(void* this, SFL_BitStream* bitStream, char priority, char reliability, char orderingChannel)", sendPacket, 6)

function sendPacket(this, bitStream, priority, reliability, orderingChannel)
    print("packet id", bitStream.data[0])
    local res = sendPacket(this, bitStream, priority, reliability, orderingChannel)
    return res
end

На текущий момент программный интерфейс поддерживает работу с наиболее востребованными версиями клиента, а именно: 0.3.7-R1, 0.3.7-R3-1 и 0.3.7-R5-1.

Установка библиотеки производится стандартным способом: вам необходимо перенести папку с названием «sampapi» непосредственно в каталог «moonloader/lib», расположенный в корневой директории вашей игры.

Официальный репозиторий проекта, где можно отслеживать изменения и изучать исходный код, доступен по следующей ссылке: https://github.com/imring/SAMP-API.lua

Ниже представлен практический пример использования библиотеки в виде скрипта «Dialog hider». Данный пример демонстрирует принцип управления активностью диалогового окна через обращение к соответствующим структурам памяти.

Lua:
local ffi = require 'ffi'
require 'moonloader'

local sampapi = require 'sampapi'
local dialog = sampapi.require('CDialog', true)

function main()
    if sampapi.GetBase() == 0 or sampapi.GetSAMPVersion() == ffi.C.SAMP_VERSION_UNKNOWN then
        return
    end

    while true do
        wait(0)

        if isKeyJustPressed(VK_HOME) then
            local ref = dialog.RefDialog()
            if ref then
                ref.m_bIsActive = 1
            end
        elseif isKeyJustPressed(VK_END) then
            local ref = dialog.RefDialog()
            if ref then
                ref.m_bIsActive = 0
            end
        end
    end
end

STRINGS.lua — Новые методы строк в Lua

Приветствую! Хочу представить вам свою первую библиотеку. Её основная задача — расширение стандартного набора методов для работы со строковыми типами данных, так как в стандартном Lua на этот счёт всё довольно скудно (string.sub, gsub, gmatch, find и так далее). Лично мне порой часто не хватало тех или иных возможностей, которыми могут похвастаться другие языки программирования. Давайте перейдём ближе к делу.

Подключение библиотеки
Для начала работы с библиотекой вам понадобится лишь подключить её в начале своего скрипта:
require "strings"

Готово, библиотека подключена, и новые методы для строк уже доступны. Работать с ними вы можете двумя способами:
1) Вызывая из класса string (например: string.count(text, ...));
2) Вызывая через двоеточие (например: text:count(...)).

Полный список методов:

1. string.insert(str, implant [, pos]) — Вставляет текст в строку на указанную позицию.
2. string.extract(str, pattern) — Вырезает указанный текст из строки.
3. string.array(str) — Преобразует строку в массив.
4. string.isEmpty(str) — Проверяет, пустая ли строка или состоит только из пробелов.
5. string.isDigit(str) — Проверяет, состоит ли строка только из цифр.
6. string.isAlpha(str) — Проверяет, состоит ли строка только из букв.
7. string.split(str [, plain]) — Разбивает строку по указанному паттерну.
8. string.lower(str) — Стандартная функция Lua, но с полной поддержкой кириллицы.
9. string.upper(str) — Стандартная функция Lua, но с полной поддержкой кириллицы.
10. string.isSpace(str) — Проверяет, состоит ли строка из невидимых символов (пробелы, управляющие символы).
11. string.isUpper(str) — Проверяет, написана ли строка в верхнем регистре.
12. string.isLower(str) — Проверяет, написана ли строка в нижнем регистре.
13. string.isSimilar(str) — Проверяет, одинаковые ли строки.
14. string.isTitle(str) — Проверяет, что первая буква в строке заглавная.
15. string.startsWith(str, prefix) — Проверяет, начинается ли строка с указанного префикса.
16. string.endsWith(str, suffix) — Проверяет, заканчивается ли строка указанным суффиксом.
17. string.capitalize(str) — Делает первую букву в строке заглавной.
18. string.tabsToSpace(str [, count]) — Преобразует табуляции в указанное количество пробелов.
19. string.spaceToTabs(str [, count]) — Преобразует указанное количество пробелов подряд в табуляцию.
20. string.center(str, width [, char]) — Заполняет строку символами (по умолчанию пробел) и выравнивает в них текст по центру.
21. string.count(str, search [, start [, end]]) — Считает количество совпадений в строке.
22. string.trim(str) — Обрезает пробелы в начале и в конце строки.
23. string.trimEnd(str) — Обрезает пробелы в конце строки.
24. string.trimStart(str) — Обрезает пробелы в начале строки.
25. string.swapCase(str) — Меняет регистр всех символов в строке на противоположный.
26. string.splitEqually(str, width) — Делит строку на N одинаковых частей.
27. string.rFind(str, pattern [, pos [, plain]]) — Аналог string.find, но поиск начинается с конца строки.
28. string.wrap(str, width) — Переносит строку (добавляет \n), если она достигла указанной длины.
29. string.levDist(str) — Вычисляет расстояние Дамерау-Левенштейна.
30. string.getSimilarity(str, str1) — Возвращает уровень сходства между строками от 0 до 1 (где 0 — абсолютно разные строки, а 1 — абсолютно одинаковые).
31. string.empty() — Возвращает пустую строку.
32. string.toCamel(str) — Возвращает текст в ВеРбЛюЖеМ РеГиСтРе.
33. string.shuffle(str [, seed]) — Перемешивает все символы в строке в случайном порядке.
34. string.reg_escape(str) — Экранирует regex-символы в строке.
35. string.cutLimit(str, max_len [, ending]) — Обрезает строку с конца по указанному количеству символов.
36. string.switchLayout(str) — Заменяет раскладку текста с RU на EN или наоборот.

Пользователь chapo предложил «лайфхак», позволяющий внедрить код библиотеки прямо в скрипт, чтобы не использовать отдельный файл:

-->> str lib (https://www.blast.hk/threads/147968/) <<--
local mt = getmetatable("String")
function mt.__index:insert(implant, pos)
    if pos == nil then return self .. implant end
    return self:sub(1, pos) .. implant .. self:sub(pos + 1)
end
function mt.__index:extract(pattern)
    self = self:gsub(pattern, "")
    return self
end
function mt.__index:array()
    local array = {}
    for s in self:gmatch(".") do array[#array + 1] = s end
    return array
end
function mt.__index:isEmpty()
    return self:find("%S") == nil
end
function mt.__index:isDigit()
    return self:find("%D") == nil
end
function mt.__index:isAlpha()
    return self:find("[%d%p]") == nil
end
function mt.__index:split(sep, plain)
    assert(not sep:isEmpty(), "Empty separator")
    local result, pos = {}, 1
    repeat
        local s, f = self:find(sep or " ", pos, plain)
        result[#result + 1] = self:sub(pos, s and s - 1)
        pos = f and f + 1
    until pos == nil
    return result
end
local orig_lower = string.lower
function mt.__index:lower()
    for i = 192, 223 do self = self:gsub(string.char(i), string.char(i + 32)) end
    self = self:gsub(string.char(168), string.char(184))
    return orig_lower(self)
end
local orig_upper = string.upper
function mt.__index:upper()
    for i = 224, 255 do self = self:gsub(string.char(i), string.char(i - 32)) end
    self = self:gsub(string.char(184), string.char(168))
    return orig_upper(self)
end
function mt.__index:isSpace()
    return self:find("^[%s%c]+$") ~= nil
end
function mt.__index:isUpper()
    return self:upper() == self
end
function mt.__index:isLower()
    return self:lower() == self
end
function mt.__index:isSimilar(str)
    return self == str
end
function mt.__index:isTitle()
    local p = self:find("[A-zА-яЁё]")
    local let = self:sub(p, p)
    return let:isSimilar(let:upper())
end
function mt.__index:startsWith(str)
    return self:sub(1, #str):isSimilar(str)
end
function mt.__index:endsWith(str)
    return self:sub(#self - #str + 1, #self):isSimilar(str)
end
function mt.__index:capitalize()
    local cap = self:sub(1, 1):upper()
    self = self:gsub("^.", cap)
    return self
end
function mt.__index:tabsToSpace(count)
    local spaces = (" "):rep(count or 4)
    self = self:gsub("\t", spaces)
    return self
end
function mt.__index:spaceToTabs(count)
    local spaces = (" "):rep(count or 4)
    self = self:gsub(spaces, "\t")
    return self
end
function mt.__index:center(width, char)
    local len = width - #self
    local s = string.rep(char or " ", len)
    return s:insert(self, math.ceil(len / 2))
end
function mt.__index:count(search, p1, p2)
    assert(not search:isEmpty(), "Empty search")
    local area = self:sub(p1 or 1, p2 or #self)
    local count, pos = 0, p1 or 1
    repeat
        local s, f = area:find(search, pos, true)
        count = s and count + 1 or count
        pos = f and f + 1
    until pos == nil
    return count
end
function mt.__index:trimEnd()
    self = self:gsub("%s*$", "")
    return self
end
function mt.__index:trimStart()
    self = self:gsub("^%s*", "")
    return self
end
function mt.__index:trim()
    self = self:match("^%s*(.-)%s*$")
    return self
end
function mt.__index:swapCase()
    local result = {}
    for s in self:gmatch(".") do
        if s:isAlpha() then s = s:isLower() and s:upper() or s:lower() end
        result[#result + 1] = s
    end
    return table.concat(result)
end
function mt.__index:splitEqually(width)
    assert(width > 0, "Width less than zero")
    assert(width <= self:len(), "Width is greater than the string length")
    local result, i = {}, 1
    repeat
        if #result == 0 or #result[#result] >= width then result[#result + 1] = "" end
        result[#result] = result[#result] .. self:sub(i, i)
        i = i + 1
    until i > #self
    return result
end
function mt.__index:rFind(pattern, pos, plain)
    local i = pos or #self
    repeat
        local result = { self:find(pattern, i, plain) }
        if next(result) ~= nil then return table.unpack(result) end
        i = i - 1
    until i <= 0
    return nil
end
function mt.__index:wrap(width)
    assert(width > 0, "Width less than zero")
    assert(width < self:len(), "Width is greater than the string length")
    local pos = 1
    self = self:gsub("(%s+)()(%S+)()", function(sp, st, word, fi)
        if fi - pos > (width or 72) then
            pos = st
            return "\n" .. word
        end
    end)
    return self
end
function mt.__index:levDist(str)
    if #self == 0 then return #str
    elseif #str == 0 then return #self
    elseif self == str then return 0 end
    local cost, matrix = 0, {}
    for i = 0, #self do matrix[i] = {}; matrix[i][0] = i end
    for i = 0, #str do matrix[0][i] = i end
    for i = 1, #self, 1 do
        for j = 1, #str, 1 do
            cost = self:byte(i) == str:byte(j) and 0 or 1
            matrix[i][j] = math.min(matrix[i - 1][j] + 1, matrix[i][j - 1] + 1, matrix[i - 1][j - 1] + cost)
        end
    end
    return matrix[#self][#str]
end
function mt.__index:getSimilarity(str)
    local dist = self:levDist(str)
    return 1 - dist / math.max(#self, #str)
end

В ходе обсуждения пользователь Jora2412 столкнулся с ошибкой "Empty separator" при попытке использовать метод split("\n"). Пользователь imring пояснил, что ошибка вызвана реализацией метода isEmpty:
function mt.__index:isEmpty()
    return self:find("%S") == nil
end
Из-за того, что split проверяет разделитель через isEmpty, пробельные символы и переносы строк (\n) считаются пустыми, что вызывает срабатывание assert. В качестве «костыля» было предложено использовать [\n] вместо \n.

Пользователь meowprd предложил альтернативный метод экранирования regex-символов (reg_escape/unplain):
local text = "().%+-*?[]^$"
print(text:gsub("([^%w])", "%%%1"))
-- output: %(%)%.%%%+%-%*%?%[%]%^%$

Пользователь F0RQU1N and иронично заметил: «Юзаем регулярки, чтоб узнать, пустая ли строка — туда эту производительность ебанную!»

2. Подключение шрифта в mimgui:
Для корректного отображения иконок в графическом интерфейсе mimgui необходимо правильно инициализировать шрифт. Ниже приведена специализированная функция подключения, где аргумент fontSize определяет итоговый размер шрифта.

Lua:
local ti = require 'tabler_icons'
local imgui = require 'mimgui'

-- Функция подключения шрифта
local function loadIconicFont(fontSize)
    local config = imgui.ImFontConfig()
    config.MergeMode = true
    config.PixelSnapH = true
    local iconRanges = imgui.new.ImWchar[3](ti.min_range, ti.max_range, 0)
    imgui.GetIO().Fonts:AddFontFromMemoryCompressedBase85TTF(ti.get_font_data_base85(), fontSize, config, iconRanges)
end

imgui.OnInitialize(function()
    loadIconicFont(14) -- Установка стандартного размера 14
    imgui.GetIO().IniFilename = nil
end)

local ti = require 'tabler_icons'

4. Пример с загрузкой шрифтов разных размеров (автор: CaJlaT):
Если в вашем скрипте требуется использовать иконки разного масштаба одновременно, воспользуйтесь следующим расширенным методом загрузки.

Lua:
function loadFonts(sizes)
    local fonts = {}
    local config = imgui.ImFontConfig()
    config.MergeMode = true
    config.PixelSnapH = true
    local iconRanges = imgui.new.ImWchar[3](ti.min_range, ti.max_range, 0)
    
    -- Обязательная базовая инициализация для корректного слияния:
    imgui.GetIO().Fonts:AddFontFromMemoryCompressedBase85TTF(ti.get_font_data_base85(), 14, config, iconRanges)
    
    for i, v in ipairs(sizes) do
        -- Подгрузка основного шрифта (например, Trebuchet MS Bold) и слияние его с иконками Tabler Icons:
        fonts[v] = imgui.GetIO().Fonts:AddFontFromFileTTF(getFolderPath(0x14) .. '\\trebucbd.ttf', v, nil, imgui.GetIO().Fonts:GetGlyphRangesCyrillic())
        imgui.GetIO().Fonts:AddFontFromMemoryCompressedBase85TTF(ti.get_font_data_base85(), v, config, iconRanges)
    end
    return fonts
end

imgui.OnInitialize(function()
    fonts = loadFonts({16, 25})
    imgui.GetIO().IniFilename = nil
end)

-- Пример использования в цикле OnFrame:
imgui.PushFont(fonts[16])
imgui.Text(ti'info-circle' .. u8' Этот текст размером 16')
imgui.PopFont()

imgui.PushFont(fonts[25])
imgui.Text(ti'info-circle' .. u8' Этот текст размером 25')
imgui.PopFont()
